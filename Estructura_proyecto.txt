Estructura del proyecto

  CLASES BASES

  Clase Carta

Atributos:
  color : string
  valor : string
  tipo : string
Métodos:
  MostrarCarta()
  GetValorNumerico() → virtual (dependiendo del tipo de carta)

            Clase Jugador
Atributos:
  nombre : string
  idJugador : int
  mano : List<Carta>
  puntos : int
Métodos:
  RecibirCarta(Carta c)
  MostrarMano()
  TomarDecision() = abstracto 
  JugarCarta() (solo se usa en UNO)

            Clase Juego
Atributos:
  nombreJuego : string
  jugadores : List<Jugador>
  mazo : List<Carta>
  descarte : List<Carta>
  turnoActual : int
Métodos:
  IniciarJuego()
  RepartirCartas()
  SiguienteTurno()
  MostrarEstado()
  DeterminarGanador()

			UNO 

	Clase CartaUNO
Atributos:

esEspecial : bool
efecto : string (para cartas como +2, +4, reversa, bloqueo, cambioColor)

Métodos:
AplicarEfecto(JuegoUNO juego, JugadorUNO siguiente)


	Clase JugadorUNO (hereda de Jugador)

Métodos:
TomarDecision()
ObtenerCartasValidas()
JugarCartaValida()


	Clase JugadorAleatorio (hereda de JugadorUNO)

Métodos:
TomarDecision()

	Clase JugadorCalculador (hereda de JugadorUNO)

Métodos:
EvaluarSiguienteJugador()
TomarDecision() 

	Clase JuegoUNO (hereda de Juego)

Atributos:
sentidoHorario : bool

Métodos:
JugarTurno()
AplicarEfectoCarta()
VerificarGanador()

BLACKJACK

Clase CartaPoker (hereda de Carta)

Atributos:
figura : string (espadas, corazones, diamantes o tréboles)

Métodos:
GetValorNumerico() → devuelve puntaje (2–10, J/Q/K = 10, As = 1 u 11)
Clase JugadorBlackjack (hereda de Jugador)

Métodos:
CalcularPuntaje()
TomarDecision() (depende del tipo de jugador)

Clase JugadorCauteloso (hereda de JugadorBlackjack)

Atributos:
limitePuntos : int

Métodos:
TomarDecision() → pide carta hasta alcanzar o superar el límite

Clase JugadorTemerario (hereda de JugadorBlackjack)

Métodos:
TomarDecision() → pide cartas hasta sacar 21 o pasarse

Clase Dealer (hereda de JugadorBlackjack)

Métodos:
BarajarMazo()
RepartirCartas()
TomarDecision() → saca cartas hasta tener 17 o más puntos

Clase JuegoBlackjack (hereda de Juego)

Atributos:
dealer : Dealer
rondas : int

Métodos:
JugarRonda()
CalcularPuntos(Jugador j)
CompararConDealer()
